# Kreiranje baze podataka
CREATE DATABASE IF NOT EXISTS db_vozila CHARACTER SET utf8mb4 
COLLATE utf8mb4_unicode_ci;
# chcp 65001 - da bi se videla slova srpske latinice
# TRUNCATE TABLE ime_tabele; - prazni tabelu i resetuje auto_increment na 1
# kreiranje tabela
CREATE TABLE IF NOT EXISTS vlasnik_vozila(
vlasnik_id INT AUTO_INCREMENT,
vlasnik_ime VARCHAR(50) NOT NULL,
vlasnik_prezime VARCHAR(60) NOT NULL,
adresa VARCHAR(100),
mesto VARCHAR(60),
PRIMARY KEY(vlasnik_id)
)ENGINE=INNODB;

# tabela sa stranim ključem
CREATE TABLE IF NOT EXISTS vozila(
vozila_id INT AUTO_INCREMENT,
marka VARCHAR(40) NOT NULL,
model VARCHAR(60) NOT NULL,
godiste INT NOT NULL,
motor_kw DECIMAL(5,1) NOT NULL,
datum_reg DATE,
reg_oznaka VARCHAR(10),
vlasnik_id INT NOT NULL,
CONSTRAINT fk_vlasnik
FOREIGN KEY vlasnik_vozila_fk(vlasnik_id)
REFERENCES vlasnik_vozila(vlasnik_id)
ON UPDATE CASCADE
ON DELETE CASCADE,
PRIMARY KEY(vozila_id)
)ENGINE=INNODB;

ALTER TABLE `vozila` ADD UNIQUE(`reg_oznaka`);
